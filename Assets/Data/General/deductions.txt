# Deductions setup
#
# [ID]
# position: X,Y (The node's position)
# uiSize: 1 (The node's size factor)
# textPosition: Left/Right/Top/Bottom (Where should the text be attached)
# isVisible: True (True if the node is visible by default)
# linkedNode: ID (The linked node's ID)
# isActiveLink: True (True if active in a switch)
# isEndNode: True (True if this is the end node)
# endScript: Path (The end script to load if this node is selected)
# prerequistes: Node1, Node2, ... (The list of nodes that are required to make this one visible)

# Exemple : 
# [Deduction_Test1]
# position: 1,0
# textPosition: Left
# isVisible: true
